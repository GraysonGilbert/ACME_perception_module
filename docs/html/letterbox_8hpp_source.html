57inline LetterboxMeta letterbox(const cv::Mat& src, cv::Mat& destination,
58 const cv::Size& destination_size,
59 const cv::Scalar& pad_value = {114, 114, 114}) {
60 LetterboxMeta metadata;
61 metadata.source_size = src.size();
62 metadata.destination_size = destination_size;
63 float ratio = std::min(destination_size.width / (float)src.cols,
64 destination_size.height / (float)src.rows);
65 metadata.scale = ratio;
66
67 int new_width = int(std::round(src.cols * ratio));
68 int new_height = int(std::round(src.rows * ratio));
69 cv::Mat resized;
70 cv::resize(src, resized, cv::Size(new_width, new_height), 0, 0,
71 cv::INTER_LINEAR);
72
73 int destination_width = destination_size.width - new_width;
74 int destination_height = destination_size.height - new_height;
75 metadata.pad_left = destination_width / 2;
76 metadata.pad_right = destination_width - metadata.pad_left;
77 metadata.pad_top = destination_height / 2;
78 metadata.pad_bottom = destination_height - metadata.pad_top;
79
80 cv::copyMakeBorder(resized, destination, metadata.pad_top,
81 metadata.pad_bottom, metadata.pad_left, metadata.pad_right,
82 cv::BORDER_CONSTANT, pad_value);
83 return metadata;
84}

93inline cv::Rect2f unletterboxBox(const cv::Rect2f& box,
94 const LetterboxMeta& metadata) {
95 // box is in destination (letterboxed) coords; map back to original image
96 // coords
97 float x = (box.x - metadata.pad_left) / metadata.scale;
98 float y = (box.y - metadata.pad_top) / metadata.scale;
99 float w = box.width / metadata.scale;
100 float h = box.height / metadata.scale;
101 return {x, y, w, h};
102}

111inline cv::Mat unletterboxDepth(const cv::Mat& depth_destination,
112 const LetterboxMeta& metadata) {
113 // depthdestination is HxW (destination_size) CV_32F; remove padding then
114 // resize back
115 cv::Rect region_of_interest(metadata.pad_left, metadata.pad_top,
116 metadata.destination_size.width -
117 (metadata.pad_left + metadata.pad_right),
118 metadata.destination_size.height -
119 (metadata.pad_top + metadata.pad_bottom));
120 cv::Mat cropped = depth_destination(region_of_interest);
121 cv::Mat restored;
122 cv::resize(cropped, restored, metadata.source_size, 0, 0, cv::INTER_CUBIC);
123 return restored;
124}
